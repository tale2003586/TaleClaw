\sectionTitle{项目经历}{\faCode}
%======================================================================
\begin{itemize}
\item \textbf{TaleClaw Agent Runtime / 多层编排 Coding Agent（面向真实任务的 Agent 执行平台）}
\begin{itemize}
\item \textbf{项目概述}: 独立实现统一 Agent Runtime，支撑 Web、CLI、Telegram 等入口复用会话管理、模型路由、工具执行、插件扩展、记忆和运行报告；在其上构建面向真实代码仓库的 Coding Agent，支持“读仓库 $\rightarrow$ 改实现 $\rightarrow$ 跑验证 $\rightarrow$ 输出报告”的闭环，形成约 44K 行主体应用代码、59K 行核心源码和 110 个测试/评测文件。
\item \textbf{工具调用治理}: 面对工具、插件调用规则容易堆进 prompt、运行过程难以稳定扩展、工具边界难以规范管理的问题，将一次执行拆分为上下文构建、模型调用、工具执行、结果写回和收尾报告等阶段，并在关键节点接入工具权限、工作区路径校验和 hook 机制，使工具治理从“模型自觉遵守”转为运行时约束。
\item \textbf{可观测链路}: 面对长任务失败后难以复盘的问题，为每次运行记录 trace、metrics、context metrics、report 和 workspace diff，将上下文构建、模型调用、工具调用、失败原因、token、耗时与压缩状态沉淀为结构化证据，支持定位“哪一步失败、为什么失败、上下文是否被压缩”。
\item \textbf{任务会话与多智能体编排}: 为每个 coding 请求创建独立 task session；主 Agent 负责规划与汇总，subagent 作为有任务边界的一次性执行体，teammate 负责持续化协作解决复杂问题；通过交流协议、失败处理、retry hint 和任务收窄规则并行推进任务，使失败能被消费而不是被反复重演。
\item \textbf{Coding 上下文特化重构}: 在通用分 section 预算、模型调用前 token 超限检测、类型化工具结果压缩和 tool result store 的基础上，面向代码仓库任务将送入模型的上下文拆为系统前缀、任务状态和最近消息三部分；其中任务状态由工具结果、子任务返回、上下文压缩和中断事件驱动状态转移，使后续 run 能基于 checkpoint 进行中断恢复。
\end{itemize}

\item \textbf{代码安全 RAG（面向代码审查与修复建议的证据检索系统）}
\begin{itemize}
    \item \textbf{知识库构建}: 面对 Coding Agent 在安全问题上容易只依赖模型常识的问题，基于约 6GB 安全资料构建本地安全知识库，为代码审查和修复建议提供可追溯的漏洞依据和参考证据。
    \item \textbf{结构化切片}: 面对固定窗口切片会破坏漏洞公告和规则语义的问题，分别为安全公告、Semgrep 规则和 Markdown 文档设计结构化切片方式，并保留 CVE、CWE、severity、package、rule id、language 等元数据，使检索结果能回到明确来源。
    \item \textbf{检索路由}: 面对 query rewrite 可能改写漂移、错误触发 RAG 的问题，重构检索路由为“先召回、再按证据置信度决策”的流程：高置信度直用，中置信度轻量扩展，低置信度再进行 LLM rewrite 或多查询分解，并对强标识符问题保留原文候选。
    \item \textbf{评测闭环}: 面对 RAG 优化缺少数据依据的问题，建立 250 条用户问题的 matrix / ablation 评测，按问题类型观察 p@1、MRR、latency 等指标；同时将检索决策、命中证据和注入内容写入 trace，用于区分问题来自召回、排序还是路由。
\end{itemize}

\end{itemize}
